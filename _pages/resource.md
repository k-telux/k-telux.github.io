/* slide filter: matches title, description, folder and journal; folders without matches hide, others open */
(() => {
  const root = document.querySelector('[data-fm]');
  const input = root && root.querySelector('[data-fm-filter]');
  if (!input) return;
  const status = root.querySelector('[data-fm-status]');
  const empty = root.querySelector('[data-fm-empty]');
  const folders = [...root.querySelectorAll('.fm-folder')].map((folder) => ({
    folder, open: folder.open,
    files: [...folder.querySelectorAll('.fm-file')].map((file) => ({ file, text: (folder.querySelector('summary').textContent + ' ' + file.textContent).toLowerCase() })),
  }));
  const total = folders.reduce((n, f) => n + f.files.length, 0);
  input.addEventListener('input', () => {
    const words = input.value.toLowerCase().split(/\s+/).filter(Boolean);
    let shown = 0;
    folders.forEach((f) => {
      let hits = 0;
      f.files.forEach(({ file, text }) => { const hit = words.every((w) => text.includes(w)); file.hidden = !hit; hits += hit; });
      f.folder.hidden = !hits;
      if (words.length && hits) f.folder.open = true;
      shown += hits;
    });
    empty.hidden = shown > 0;
    status.textContent = words.length ? shown + ' of ' + total + ' decks' : '';
  });
})();
</script>
